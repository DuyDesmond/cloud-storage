-- ============================================================================
-- 1. FOLDERS TABLE
-- ============================================================================
CREATE TABLE folders (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    owner_id UUID NOT NULL REFERENCES users(id) ON DELETE CASCADE,
    parent_folder_id UUID REFERENCES folders(id) ON DELETE CASCADE,
    path LTREE NOT NULL,
    folder_name TEXT NOT NULL,
    is_trashed BOOLEAN NOT NULL DEFAULT FALSE,
    trashed_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),

    CONSTRAINT check_folder_no_self_parent CHECK (id != parent_folder_id),
    CONSTRAINT check_folder_name_length CHECK (char_length(folder_name) BETWEEN 1 AND 255),
    CONSTRAINT check_folder_trash_state CHECK (
        (is_trashed = FALSE AND trashed_at IS NULL) OR
        (is_trashed = TRUE AND trashed_at IS NOT NULL)
    )
);

CREATE INDEX idx_folders_owner ON folders (owner_id);
CREATE INDEX idx_folders_parent ON folders (parent_folder_id);
CREATE INDEX idx_folders_path_gist ON folders USING GIST (path);
CREATE INDEX idx_folders_trashed_at ON folders (trashed_at) WHERE is_trashed = TRUE;

-- NOTE: intentionally no unique index on (parent_folder_id, folder_name) -- folders are allowed
-- to share names per product rules.

-- ============================================================================
-- 2. FILES TABLE

    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),

    is_trashed BOOLEAN NOT NULL DEFAULT FALSE,

    created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
    updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),

    CONSTRAINT check_positive_size CHECK (size_bytes >= 0),
    CONSTRAINT check_file_name_length CHECK (char_length(file_name) BETWEEN 1 AND 255),

        (is_trashed = TRUE AND trashed_at IS NOT NULL)

ON files (parent_folder_id, file_name)
WHERE is_trashed = FALSE AND parent_folder_id IS NOT NULL;

-- Enforce unique active file names in user root directory
CREATE UNIQUE INDEX idx_unique_active_file_in_root
ON files (owner_id, file_name)

RETURNS TRIGGER AS $$
BEGIN
    NEW.updated_at = NOW();
    RETURN NEW;

-- ============================================================================
-- 5b. PATH LABEL HELPER
-- ============================================================================
-- Converts a UUID into a single ltree label (hyphens aren't valid in ltree labels).
-- Shared by the INSERT-time path setter and the move (reparent) triggers below,
-- so the label format only lives in one place.
CREATE OR REPLACE FUNCTION uuid_ltree_label(p_id UUID)
RETURNS LTREE AS $$
    SELECT text2ltree(replace(p_id::text, '-', '_'));
$$ LANGUAGE sql IMMUTABLE;

-- ============================================================================
-- 5c. PATH ASSIGNMENT ON INSERT
-- ============================================================================
-- Computes `path` automatically on creation -- the app never supplies it directly.
-- Root-level items get a single-label path from their own id; nested items get
-- their parent's path plus their own id appended.
CREATE OR REPLACE FUNCTION set_folder_path_on_insert()
RETURNS TRIGGER AS $$
DECLARE
    parent_path LTREE;
BEGIN
    IF NEW.parent_folder_id IS NULL THEN
        NEW.path := uuid_ltree_label(NEW.id);
    ELSE
        SELECT path INTO parent_path FROM folders WHERE id = NEW.parent_folder_id;
        IF parent_path IS NULL THEN
            RAISE EXCEPTION 'Parent folder % does not exist', NEW.parent_folder_id;
        END IF;
        NEW.path := parent_path || uuid_ltree_label(NEW.id);
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_set_folder_path_on_insert
BEFORE INSERT ON folders
FOR EACH ROW EXECUTE FUNCTION set_folder_path_on_insert();

CREATE OR REPLACE FUNCTION set_file_path_on_insert()
RETURNS TRIGGER AS $$
DECLARE
    parent_path LTREE;
BEGIN
    IF NEW.parent_folder_id IS NULL THEN
        NEW.path := uuid_ltree_label(NEW.id);
    ELSE
        SELECT path INTO parent_path FROM folders WHERE id = NEW.parent_folder_id;
        IF parent_path IS NULL THEN
            RAISE EXCEPTION 'Parent folder % does not exist', NEW.parent_folder_id;
        END IF;
        NEW.path := parent_path || uuid_ltree_label(NEW.id);
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_set_file_path_on_insert
BEFORE INSERT ON files
FOR EACH ROW EXECUTE FUNCTION set_file_path_on_insert();

-- NOTE: these fire BEFORE INSERT, so NEW.id is already populated (the id DEFAULT
-- gen_random_uuid() resolves before triggers see the row) but NEW.path needs no
-- default of its own -- the app should never set it directly on insert.

-- ============================================================================
-- 6. CYCLE PREVENTION ON FOLDER MOVE
-- ============================================================================
-- Prevents moving a folder into one of its own descendants (or itself).
CREATE OR REPLACE FUNCTION prevent_folder_cycle()
RETURNS TRIGGER AS $$
DECLARE
    new_parent_path LTREE;
BEGIN
    IF NEW.parent_folder_id IS NULL OR NEW.parent_folder_id = OLD.parent_folder_id THEN
        RETURN NEW;
    END IF;

    SELECT path INTO new_parent_path FROM folders WHERE id = NEW.parent_folder_id;

    IF new_parent_path IS NULL THEN
        RAISE EXCEPTION 'Target parent folder % does not exist', NEW.parent_folder_id;
    END IF;

    IF new_parent_path <@ OLD.path THEN
        RAISE EXCEPTION 'Cannot move folder % into its own descendant %', OLD.id, NEW.parent_folder_id;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_prevent_folder_cycle
BEFORE UPDATE OF parent_folder_id ON folders
FOR EACH ROW EXECUTE FUNCTION prevent_folder_cycle();

-- ============================================================================
-- 7. PATH MAINTENANCE ON MOVE
-- ============================================================================
-- Folders: recompute this folder's own path, then cascade the new prefix to every
-- descendant folder/file. Runs AFTER UPDATE since it needs both OLD.path (to match
-- descendants) and the already-applied NEW.path.
CREATE OR REPLACE FUNCTION reparent_folder_path()
RETURNS TRIGGER AS $$
DECLARE
    new_parent_path LTREE;
    computed_path LTREE;
BEGIN
    IF NEW.parent_folder_id IS NOT DISTINCT FROM OLD.parent_folder_id THEN
        RETURN NEW;
    END IF;

    IF NEW.parent_folder_id IS NULL THEN
        computed_path := uuid_ltree_label(NEW.id);
    ELSE
        SELECT path INTO new_parent_path FROM folders WHERE id = NEW.parent_folder_id;
        computed_path := new_parent_path || subpath(OLD.path, -1);
    END IF;

    -- Cascade prefix change to all descendants BEFORE updating this row's own path,
    -- so the WHERE path <@ OLD.path match still includes this folder's old subtree.
    UPDATE folders
        SET path = computed_path || subpath(path, nlevel(OLD.path))
        WHERE path <@ OLD.path AND id != NEW.id;

    UPDATE files
        SET path = computed_path || subpath(path, nlevel(OLD.path))
        WHERE path <@ OLD.path;

    UPDATE folders SET path = computed_path WHERE id = NEW.id;

    RETURN NULL; -- row already updated above; AFTER trigger return value is ignored anyway
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_reparent_folder_path
AFTER UPDATE OF parent_folder_id ON folders
FOR EACH ROW EXECUTE FUNCTION reparent_folder_path();

-- Files: no descendants, just recompute this row's own path on move.
CREATE OR REPLACE FUNCTION reparent_file_path()
RETURNS TRIGGER AS $$
DECLARE
    new_parent_path LTREE;
BEGIN
    IF NEW.parent_folder_id IS NOT DISTINCT FROM OLD.parent_folder_id THEN
        RETURN NEW;
    END IF;

    IF NEW.parent_folder_id IS NULL THEN
        NEW.path := uuid_ltree_label(NEW.id);
    ELSE
        SELECT path INTO new_parent_path FROM folders WHERE id = NEW.parent_folder_id;
        NEW.path := new_parent_path || subpath(NEW.path, -1);
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_reparent_file_path
BEFORE UPDATE OF parent_folder_id ON files
FOR EACH ROW EXECUTE FUNCTION reparent_file_path();

-- NOTE: for very large subtrees, the synchronous UPDATE ... WHERE path <@ OLD.path in
-- reparent_folder_path() can hold long locks. If that becomes a problem in practice,
-- switch to marking the subtree "path dirty" here and reconciling it via an async job,
-- with reads falling back to an ancestor-walk until reconciliation completes.

-- ============================================================================
-- 8. TRASH CASCADE (trash and restore)
-- ============================================================================
-- Trashing a folder cascades is_trashed/trashed_at to every descendant folder/file.
-- Restoring a folder unconditionally cascades restore to every descendant, including
-- items that were individually trashed before the ancestor was trashed at all -- e.g.
-- file X deleted on Monday, its parent folder deleted on Friday, then parent restored
-- the following Monday: X comes back too, per product rule.
CREATE OR REPLACE FUNCTION cascade_folder_trash_state()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.is_trashed = TRUE AND OLD.is_trashed = FALSE THEN
        UPDATE folders
            SET is_trashed = TRUE, trashed_at = NEW.trashed_at
            WHERE path <@ OLD.path AND id != NEW.id AND is_trashed = FALSE;

        UPDATE files
            SET is_trashed = TRUE, trashed_at = NEW.trashed_at
            WHERE path <@ OLD.path AND is_trashed = FALSE;

    ELSIF NEW.is_trashed = FALSE AND OLD.is_trashed = TRUE THEN
        UPDATE folders
            SET is_trashed = FALSE, trashed_at = NULL
            WHERE path <@ NEW.path AND id != NEW.id;

        UPDATE files
            SET is_trashed = FALSE, trashed_at = NULL
            WHERE path <@ NEW.path;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_cascade_folder_trash
AFTER UPDATE OF is_trashed ON folders
FOR EACH ROW EXECUTE FUNCTION cascade_folder_trash_state();

-- App layer contract:
--   * User-initiated delete on a specific item -> app sets is_trashed=TRUE,
--     trashed_at=now() on that row directly.
--   * Cascade from an ancestor delete/restore -> handled entirely by the trigger above.
--   * Restoring an ancestor always restores its full subtree unconditionally,
--     regardless of each descendant's own trash history.

-- ============================================================================
-- 9. SHARE GRANTEE != OWNER
-- ============================================================================
-- Prevents sharing a file/folder with its own owner (redundant/confusing row).
CREATE OR REPLACE FUNCTION check_grantee_not_owner()
RETURNS TRIGGER AS $$
DECLARE
    target_owner UUID;
BEGIN
    IF NEW.file_id IS NOT NULL THEN
        SELECT owner_id INTO target_owner FROM files WHERE id = NEW.file_id;
    ELSE
        SELECT owner_id INTO target_owner FROM folders WHERE id = NEW.folder_id;
    END IF;

    IF target_owner = NEW.grantee_id THEN
        RAISE EXCEPTION 'Cannot share an item with its own owner';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_grantee_not_owner
BEFORE INSERT OR UPDATE ON direct_shares
FOR EACH ROW EXECUTE FUNCTION check_grantee_not_owner();

-- ============================================================================
-- 10. EFFECTIVE PERMISSION (public link + private share, highest wins)
-- ============================================================================
-- Computes what a given user can actually do with a given file or folder,
-- accounting for: direct shares (on the item or any ancestor folder), and
-- active public links (on the item or any ancestor folder). Public and
-- private grants are never merged/mutated on write -- this function always
-- combines them at read time, taking the highest permission found, so that
-- revoking a public link cleanly falls back to any private grant.
--
-- Returns NULL if the user has no access at all.
CREATE OR REPLACE FUNCTION effective_permission(
    p_target_path LTREE,
    p_is_file BOOLEAN,       -- TRUE if p_target_path refers to a file, FALSE for folder
    p_target_id UUID,        -- the file_id or folder_id itself, for exact (non-ancestor) share rows
    p_user_id UUID
) RETURNS permission_level AS $$
DECLARE
    result permission_level;
BEGIN
    SELECT MAX(perm)::permission_level INTO result FROM (
        -- direct share targeting an ancestor folder (or this folder itself)
        SELECT ds.permission::text AS perm
        FROM direct_shares ds
        JOIN folders f ON f.id = ds.folder_id
        WHERE ds.grantee_id = p_user_id
          AND f.is_trashed = FALSE
          AND f.path @> p_target_path

        UNION ALL

        -- direct share targeting this exact file
        SELECT ds.permission::text
        FROM direct_shares ds
        JOIN files fl ON fl.id = ds.file_id
        WHERE p_is_file AND ds.grantee_id = p_user_id AND ds.file_id = p_target_id
          AND fl.is_trashed = FALSE

        UNION ALL

        -- active public link on an ancestor folder (or this folder itself)
        SELECT psl.permission::text
        FROM public_share_links psl
        JOIN folders f ON f.id = psl.folder_id
        WHERE psl.is_active
          AND (psl.expires_at IS NULL OR psl.expires_at > now())
          AND f.is_trashed = FALSE
          AND f.path @> p_target_path

        UNION ALL

        -- active public link on this exact file
        SELECT psl.permission::text
        FROM public_share_links psl
        JOIN files fl ON fl.id = psl.file_id
        WHERE p_is_file AND psl.is_active
          AND (psl.expires_at IS NULL OR psl.expires_at > now())
          AND psl.file_id = p_target_id
          AND fl.is_trashed = FALSE
    ) AS perms(perm)
    WHERE perm = 'edit' OR perm = 'view'; -- guards against stray NULLs before the MAX/cast

    RETURN result; -- NULL if no matching row -> caller treats as "no access"
END;
$$ LANGUAGE plpgsql STABLE;

-- NOTE on MAX() over an enum: 'edit' sorts after 'view' alphabetically, which happens
-- to match "edit is higher priority than view" here, but that's incidental, not
-- guaranteed by Postgres enum ordering. Postgres enum comparison actually follows
-- declaration order (CREATE TYPE permission_level AS ENUM ('view','edit')), so 'edit' >
-- 'view' holds by declared order, not alphabetically -- just don't reorder the enum
-- values later without re-checking this function.

-- ============================================================================
-- 11. OWNER ACCESS
-- ============================================================================
-- effective_permission() above ONLY looks at direct_shares / public_share_links --
-- it knows nothing about ownership, and check_grantee_not_owner (section 9) actively
-- prevents an owner from ever having a direct_shares row of their own. So calling
-- effective_permission() for the owner of an item returns NULL (no access), which is
-- wrong: the owner always has full rights, and those rights are broader than 'edit'
-- (delete, reshare, transfer ownership, manage public links) -- broader than the
-- view/edit enum can express at all.
--
-- Ownership must be checked FIRST, before ever consulting effective_permission().
-- is_owner() is a thin helper so the app doesn't duplicate this lookup; the actual
-- "owner implies full rights" branching belongs in the app's authorization layer,
-- since permission_level has no value that means "owner".
CREATE OR REPLACE FUNCTION is_owner(
    p_is_file BOOLEAN,
    p_target_id UUID,
    p_user_id UUID
) RETURNS BOOLEAN AS $$
DECLARE
    target_owner UUID;
BEGIN
    IF p_is_file THEN
        SELECT owner_id INTO target_owner FROM files WHERE id = p_target_id;
    ELSE
        SELECT owner_id INTO target_owner FROM folders WHERE id = p_target_id;
    END IF;

    RETURN target_owner IS NOT NULL AND target_owner = p_user_id;
END;
$$ LANGUAGE plpgsql STABLE;

-- App-layer authorization contract:
--   1. Check is_owner() first -> if true, grant full rights (view, edit, delete,
--      reshare, manage public links, transfer ownership), and skip
--      effective_permission() entirely.
--   2. Otherwise, call effective_permission() and map the result (or NULL) to
--      whatever the requested action needs (e.g. NULL or 'view' -> deny edit/delete/
--      reshare requests; 'edit' -> allow view+edit but still deny delete/reshare/
--      transfer, which remain owner-only regardless of share permission).

-- ============================================================================
-- 5b. PATH LABEL HELPER
-- ============================================================================
-- Converts a UUID into a single ltree label (hyphens aren't valid in ltree labels).
-- Shared by the INSERT-time path setter and the move (reparent) triggers below,
-- so the label format only lives in one place.
CREATE OR REPLACE FUNCTION uuid_ltree_label(p_id UUID)
RETURNS LTREE AS $$
    SELECT text2ltree(replace(p_id::text, '-', '_'));
$$ LANGUAGE sql IMMUTABLE;

-- ============================================================================
-- 5c. PATH ASSIGNMENT ON INSERT
-- ============================================================================
-- Computes `path` automatically on creation -- the app never supplies it directly.
-- Root-level items get a single-label path from their own id; nested items get
-- their parent's path plus their own id appended.
CREATE OR REPLACE FUNCTION set_folder_path_on_insert()
RETURNS TRIGGER AS $$
DECLARE
    parent_path LTREE;
BEGIN
    IF NEW.parent_folder_id IS NULL THEN
        NEW.path := uuid_ltree_label(NEW.id);
    ELSE
        SELECT path INTO parent_path FROM folders WHERE id = NEW.parent_folder_id;
        IF parent_path IS NULL THEN
            RAISE EXCEPTION 'Parent folder % does not exist', NEW.parent_folder_id;
        END IF;
        NEW.path := parent_path || uuid_ltree_label(NEW.id);
    END IF;
    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_set_folder_path_on_insert
BEFORE INSERT ON folders
FOR EACH ROW EXECUTE FUNCTION set_folder_path_on_insert();

-- ============================================================================
-- 6. CYCLE PREVENTION ON FOLDER MOVE
-- ============================================================================
-- Prevents moving a folder into one of its own descendants (or itself).
CREATE OR REPLACE FUNCTION prevent_folder_cycle()
RETURNS TRIGGER AS $$
DECLARE
    new_parent_path LTREE;
BEGIN
    IF NEW.parent_folder_id IS NULL OR NEW.parent_folder_id = OLD.parent_folder_id THEN
        RETURN NEW;
    END IF;

    SELECT path INTO new_parent_path FROM folders WHERE id = NEW.parent_folder_id;

    IF new_parent_path IS NULL THEN
        RAISE EXCEPTION 'Target parent folder % does not exist', NEW.parent_folder_id;
    END IF;

    IF new_parent_path <@ OLD.path THEN
        RAISE EXCEPTION 'Cannot move folder % into its own descendant %', OLD.id, NEW.parent_folder_id;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_prevent_folder_cycle
BEFORE UPDATE OF parent_folder_id ON folders
FOR EACH ROW EXECUTE FUNCTION prevent_folder_cycle();

-- ============================================================================
-- 7. PATH MAINTENANCE ON MOVE
-- ============================================================================
-- Folders: recompute this folder's own path, then cascade the new prefix to every
-- descendant folder/file. Runs AFTER UPDATE since it needs both OLD.path (to match
-- descendants) and the already-applied NEW.path.
CREATE OR REPLACE FUNCTION reparent_folder_path()
RETURNS TRIGGER AS $$
DECLARE
    new_parent_path LTREE;
    computed_path LTREE;
BEGIN
    IF NEW.parent_folder_id IS NOT DISTINCT FROM OLD.parent_folder_id THEN
        RETURN NEW;
    END IF;

    IF NEW.parent_folder_id IS NULL THEN
        computed_path := uuid_ltree_label(NEW.id);
    ELSE
        SELECT path INTO new_parent_path FROM folders WHERE id = NEW.parent_folder_id;
        computed_path := new_parent_path || subpath(OLD.path, -1);
    END IF;

    -- Cascade prefix change to all descendants BEFORE updating this row's own path,
    -- so the WHERE path <@ OLD.path match still includes this folder's old subtree.
    UPDATE folders
        SET path = computed_path || subpath(path, nlevel(OLD.path))
        WHERE path <@ OLD.path AND id != NEW.id;

    UPDATE files
        SET path = computed_path || subpath(path, nlevel(OLD.path))
        WHERE path <@ OLD.path;

    UPDATE folders SET path = computed_path WHERE id = NEW.id;

    RETURN NULL; -- row already updated above; AFTER trigger return value is ignored anyway
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_reparent_folder_path
AFTER UPDATE OF parent_folder_id ON folders
FOR EACH ROW EXECUTE FUNCTION reparent_folder_path();

-- Files: no descendants, just recompute this row's own path on move.
CREATE OR REPLACE FUNCTION reparent_file_path()
RETURNS TRIGGER AS $$
DECLARE
    new_parent_path LTREE;
BEGIN
    IF NEW.parent_folder_id IS NOT DISTINCT FROM OLD.parent_folder_id THEN
        RETURN NEW;
    END IF;

    IF NEW.parent_folder_id IS NULL THEN
        NEW.path := uuid_ltree_label(NEW.id);
    ELSE
        SELECT path INTO new_parent_path FROM folders WHERE id = NEW.parent_folder_id;
        NEW.path := new_parent_path || subpath(NEW.path, -1);
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_reparent_file_path
BEFORE UPDATE OF parent_folder_id ON files

-- NOTE: for very large subtrees, the synchronous UPDATE ... WHERE path <@ OLD.path in
-- reparent_folder_path() can hold long locks. If that becomes a problem in practice,
-- switch to marking the subtree "path dirty" here and reconciling it via an async job,
-- with reads falling back to an ancestor-walk until reconciliation completes.

-- ============================================================================
-- 8. TRASH CASCADE (trash and restore)
-- ============================================================================
-- Trashing a folder cascades is_trashed/trashed_at to every descendant folder/file.
-- Restoring a folder unconditionally cascades restore to every descendant, including
-- items that were individually trashed before the ancestor was trashed at all -- e.g.
-- file X deleted on Monday, its parent folder deleted on Friday, then parent restored
-- the following Monday: X comes back too, per product rule.
CREATE OR REPLACE FUNCTION cascade_folder_trash_state()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.is_trashed = TRUE AND OLD.is_trashed = FALSE THEN
        UPDATE folders
            SET is_trashed = TRUE, trashed_at = NEW.trashed_at
            WHERE path <@ OLD.path AND id != NEW.id AND is_trashed = FALSE;

        UPDATE files
            SET is_trashed = TRUE, trashed_at = NEW.trashed_at
            WHERE path <@ OLD.path AND is_trashed = FALSE;

    ELSIF NEW.is_trashed = FALSE AND OLD.is_trashed = TRUE THEN
        UPDATE folders
            SET is_trashed = FALSE, trashed_at = NULL
            WHERE path <@ NEW.path AND id != NEW.id;

        UPDATE files
            SET is_trashed = FALSE, trashed_at = NULL
            WHERE path <@ NEW.path;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_cascade_folder_trash
AFTER UPDATE OF is_trashed ON folders
FOR EACH ROW EXECUTE FUNCTION cascade_folder_trash_state();

-- App layer contract:
--   * User-initiated delete on a specific item -> app sets is_trashed=TRUE,
--     trashed_at=now() on that row directly.
--   * Cascade from an ancestor delete/restore -> handled entirely by the trigger above.
--   * Restoring an ancestor always restores its full subtree unconditionally,
--     regardless of each descendant's own trash history.

-- ============================================================================
-- 9. SHARE GRANTEE != OWNER
-- ============================================================================
-- Prevents sharing a file/folder with its own owner (redundant/confusing row).
CREATE OR REPLACE FUNCTION check_grantee_not_owner()
RETURNS TRIGGER AS $$
DECLARE
    target_owner UUID;
BEGIN
    IF NEW.file_id IS NOT NULL THEN
        SELECT owner_id INTO target_owner FROM files WHERE id = NEW.file_id;
    ELSE
        SELECT owner_id INTO target_owner FROM folders WHERE id = NEW.folder_id;
    END IF;

    IF target_owner = NEW.grantee_id THEN
        RAISE EXCEPTION 'Cannot share an item with its own owner';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_grantee_not_owner
BEFORE INSERT OR UPDATE ON direct_shares
FOR EACH ROW EXECUTE FUNCTION check_grantee_not_owner();

-- ============================================================================
-- 10. EFFECTIVE PERMISSION (public link + private share, highest wins)
-- ============================================================================
-- Computes what a given user can actually do with a given file or folder,
-- accounting for: direct shares (on the item or any ancestor folder), and
-- active public links (on the item or any ancestor folder). Public and
-- private grants are never merged/mutated on write -- this function always
-- combines them at read time, taking the highest permission found, so that
-- revoking a public link cleanly falls back to any private grant.
--
-- Returns NULL if the user has no access at all.
CREATE OR REPLACE FUNCTION effective_permission(
    p_target_path LTREE,
    p_is_file BOOLEAN,       -- TRUE if p_target_path refers to a file, FALSE for folder
    p_target_id UUID,        -- the file_id or folder_id itself, for exact (non-ancestor) share rows
    p_user_id UUID
) RETURNS permission_level AS $$
DECLARE
    result permission_level;
BEGIN
    SELECT MAX(perm)::permission_level INTO result FROM (
        -- direct share targeting an ancestor folder (or this folder itself)
        SELECT ds.permission::text AS perm
        FROM direct_shares ds
        JOIN folders f ON f.id = ds.folder_id
        WHERE ds.grantee_id = p_user_id
          AND f.is_trashed = FALSE
          AND f.path @> p_target_path

        UNION ALL

        -- direct share targeting this exact file
        SELECT ds.permission::text
        FROM direct_shares ds
        WHERE p_is_file AND ds.grantee_id = p_user_id AND ds.file_id = p_target_id

        UNION ALL

        -- active public link on an ancestor folder (or this folder itself)
        SELECT psl.permission::text
        FROM public_share_links psl
        JOIN folders f ON f.id = psl.folder_id
        WHERE psl.is_active
          AND (psl.expires_at IS NULL OR psl.expires_at > now())
          AND f.is_trashed = FALSE
          AND f.path @> p_target_path

        UNION ALL

        -- active public link on this exact file
        SELECT psl.permission::text
        FROM public_share_links psl
        WHERE p_is_file AND psl.is_active
          AND (psl.expires_at IS NULL OR psl.expires_at > now())
          AND psl.file_id = p_target_id
    ) AS perms(perm)
    WHERE perm = 'edit' OR perm = 'view'; -- guards against stray NULLs before the MAX/cast

    RETURN result; -- NULL if no matching row -> caller treats as "no access"
END;
$$ LANGUAGE plpgsql STABLE;

-- NOTE on MAX() over an enum: 'edit' sorts after 'view' alphabetically, which happens
-- to match "edit is higher priority than view" here, but that's incidental, not
-- guaranteed by Postgres enum ordering. Postgres enum comparison actually follows
-- declaration order (CREATE TYPE permission_level AS ENUM ('view','edit')), so 'edit' >
-- 'view' holds by declared order, not alphabetically -- just don't reorder the enum
-- values later without re-checking this function.

-- ============================================================================
-- 11. OWNER ACCESS
-- ============================================================================
-- effective_permission() above ONLY looks at direct_shares / public_share_links --
-- it knows nothing about ownership, and check_grantee_not_owner (section 9) actively
-- prevents an owner from ever having a direct_shares row of their own. So calling
-- effective_permission() for the owner of an item returns NULL (no access), which is
-- wrong: the owner always has full rights, and those rights are broader than 'edit'
-- (delete, reshare, transfer ownership, manage public links) -- broader than the
-- view/edit enum can express at all.
--
-- Ownership must be checked FIRST, before ever consulting effective_permission().
-- is_owner() is a thin helper so the app doesn't duplicate this lookup; the actual
-- "owner implies full rights" branching belongs in the app's authorization layer,
-- since permission_level has no value that means "owner".
CREATE OR REPLACE FUNCTION is_owner(
    p_is_file BOOLEAN,
    p_target_id UUID,
    p_user_id UUID
) RETURNS BOOLEAN AS $$
DECLARE
    target_owner UUID;
BEGIN
    IF p_is_file THEN
        SELECT owner_id INTO target_owner FROM files WHERE id = p_target_id;
    ELSE
        SELECT owner_id INTO target_owner FROM folders WHERE id = p_target_id;
    END IF;

    RETURN target_owner IS NOT NULL AND target_owner = p_user_id;
END;
$$ LANGUAGE plpgsql STABLE;

-- App-layer authorization contract:
--   1. Check is_owner() first -> if true, grant full rights (view, edit, delete,
--      reshare, manage public links, transfer ownership), and skip
--      effective_permission() entirely.
--   2. Otherwise, call effective_permission() and map the result (or NULL) to
--      whatever the requested action needs (e.g. NULL or 'view' -> deny edit/delete/
--      reshare requests; 'edit' -> allow view+edit but still deny delete/reshare/
--      transfer, which remain owner-only regardless of share permission).

-- the following Monday: X comes back too, per product rule.
CREATE OR REPLACE FUNCTION cascade_folder_trash_state()
RETURNS TRIGGER AS $$
BEGIN
    IF NEW.is_trashed = TRUE AND OLD.is_trashed = FALSE THEN
        UPDATE folders
            SET is_trashed = TRUE, trashed_at = NEW.trashed_at
            WHERE path <@ OLD.path AND id != NEW.id AND is_trashed = FALSE;

        UPDATE files
            SET is_trashed = TRUE, trashed_at = NEW.trashed_at
            WHERE path <@ OLD.path AND is_trashed = FALSE;

    ELSIF NEW.is_trashed = FALSE AND OLD.is_trashed = TRUE THEN
        UPDATE folders
            SET is_trashed = FALSE, trashed_at = NULL
            WHERE path <@ NEW.path AND id != NEW.id;

        UPDATE files
            SET is_trashed = FALSE, trashed_at = NULL
            WHERE path <@ NEW.path;
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_cascade_folder_trash
AFTER UPDATE OF is_trashed ON folders
FOR EACH ROW EXECUTE FUNCTION cascade_folder_trash_state();

-- App layer contract:
--   * User-initiated delete on a specific item -> app sets is_trashed=TRUE,
--     trashed_at=now() on that row directly.
--   * Cascade from an ancestor delete/restore -> handled entirely by the trigger above.
--   * Restoring an ancestor always restores its full subtree unconditionally,
--     regardless of each descendant's own trash history.

-- ============================================================================
-- 9. SHARE GRANTEE != OWNER
-- ============================================================================
-- Prevents sharing a file/folder with its own owner (redundant/confusing row).
CREATE OR REPLACE FUNCTION check_grantee_not_owner()
RETURNS TRIGGER AS $$
DECLARE
    target_owner UUID;
BEGIN
    IF NEW.file_id IS NOT NULL THEN
        SELECT owner_id INTO target_owner FROM files WHERE id = NEW.file_id;
    ELSE
        SELECT owner_id INTO target_owner FROM folders WHERE id = NEW.folder_id;
    END IF;

    IF target_owner = NEW.grantee_id THEN
        RAISE EXCEPTION 'Cannot share an item with its own owner';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_grantee_not_owner
BEFORE INSERT OR UPDATE ON direct_shares
FOR EACH ROW EXECUTE FUNCTION check_grantee_not_owner();

-- ============================================================================
-- 10. EFFECTIVE PERMISSION (public link + private share, highest wins)
-- ============================================================================
-- Computes what a given user can actually do with a given file or folder,
-- accounting for: direct shares (on the item or any ancestor folder), and
-- active public links (on the item or any ancestor folder). Public and
-- private grants are never merged/mutated on write -- this function always
-- combines them at read time, taking the highest permission found, so that
-- revoking a public link cleanly falls back to any private grant.
--
-- Returns NULL if the user has no access at all.
CREATE OR REPLACE FUNCTION effective_permission(
    p_target_path LTREE,
    p_is_file BOOLEAN,       -- TRUE if p_target_path refers to a file, FALSE for folder
    p_target_id UUID,        -- the file_id or folder_id itself, for exact (non-ancestor) share rows
    p_user_id UUID
) RETURNS permission_level AS $$
DECLARE
    result permission_level;
BEGIN
    SELECT MAX(perm)::permission_level INTO result FROM (
        -- direct share targeting an ancestor folder (or this folder itself)
        SELECT ds.permission::text AS perm
        FROM direct_shares ds
        JOIN folders f ON f.id = ds.folder_id
        WHERE ds.grantee_id = p_user_id
          AND f.is_trashed = FALSE
          AND f.path @> p_target_path

        UNION ALL

        -- direct share targeting this exact file
        SELECT ds.permission::text
        FROM direct_shares ds
        WHERE p_is_file AND ds.grantee_id = p_user_id AND ds.file_id = p_target_id

        UNION ALL

        -- active public link on an ancestor folder (or this folder itself)
        SELECT psl.permission::text
        FROM public_share_links psl
        JOIN folders f ON f.id = psl.folder_id
        WHERE psl.is_active
          AND (psl.expires_at IS NULL OR psl.expires_at > now())
          AND f.is_trashed = FALSE
          AND f.path @> p_target_path

        UNION ALL

        -- active public link on this exact file
        SELECT psl.permission::text
        FROM public_share_links psl
        WHERE p_is_file AND psl.is_active
          AND (psl.expires_at IS NULL OR psl.expires_at > now())
          AND psl.file_id = p_target_id
    ) AS perms(perm)
    WHERE perm = 'edit' OR perm = 'view'; -- guards against stray NULLs before the MAX/cast

    RETURN result; -- NULL if no matching row -> caller treats as "no access"
END;
$$ LANGUAGE plpgsql STABLE;

-- NOTE on MAX() over an enum: 'edit' sorts after 'view' alphabetically, which happens
-- to match "edit is higher priority than view" here, but that's incidental, not
-- guaranteed by Postgres enum ordering. Postgres enum comparison actually follows
-- declaration order (CREATE TYPE permission_level AS ENUM ('view','edit')), so 'edit' >
-- 'view' holds by declared order, not alphabetically -- just don't reorder the enum
-- values later without re-checking this function.
